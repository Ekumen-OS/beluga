153  [[nodiscard]] std::optional<LandmarkBearing3> find_closest_bearing_landmark(
154  const LandmarkBearing3& detection_bearing_in_sensor,
155  const LandmarkCategory& detection_category,
156  const world_pose_type& sensor_pose_in_world) const {
157  const auto it = category_indices_.find(detection_category);
158  if (it == category_indices_.end()) {
159  return std::nullopt;
160  }
161  const auto& same_category_landmarks = it->second->cloud.pts;
162 
163  // find the landmark that minimizes the bearing error
164  const auto world_in_sensor_transform = sensor_pose_in_world.inverse();
165 
166  // This whole search thing is very expensive, with objects getting created, normalized and
167  // destroyed multiple times and the same transformations being calculated over and over again.
168  // This will only work as a proof-of-concept, but it needs to be optimized for large numbers of
169  // landmarks.
170  const auto minimization_function = [&detection_bearing_in_sensor, &world_in_sensor_transform](
171  const auto& landmark_a_position_in_world,
172  const auto& landmark_b_position_in_world) {
173  // convert the landmark locations relative to the sensor frame
174  const auto landmark_a_bearing_in_sensor = (world_in_sensor_transform * landmark_a_position_in_world).normalized();
175  const auto landmark_b_bearing_in_sensor = (world_in_sensor_transform * landmark_b_position_in_world).normalized();
176 
177  // find the landmark that minimizes the bearing error by maximizing the dot product against the
178  // detection bearing vector
179  const auto dot_product_a = landmark_a_bearing_in_sensor.dot(detection_bearing_in_sensor);
180  const auto dot_product_b = landmark_b_bearing_in_sensor.dot(detection_bearing_in_sensor);
181 
182  return dot_product_a > dot_product_b;
183  };
184 
185  auto min = std::min_element(same_category_landmarks.begin(), same_category_landmarks.end(), minimization_function);
186 
187  if (min == same_category_landmarks.end()) {
188  return std::nullopt;
189  }
190 
191  // find the normalized bearing vector to the landmark, relative to the sensor frame
192  const auto landmark_position_in_sensor = world_in_sensor_transform * *min;
193  return landmark_position_in_sensor.normalized();
194  }

229  std::unordered_map<LandmarkCategory, std::size_t> category_sizes;
230  for (const auto& l : landmarks) {
