120  template <class StateRange>
121  void prepare(const measurement_type& points, StateRange&& states) {
122  // The mask is always sized to match the measurement, so that `operator()` can consume it
123  // unconditionally: when beam skipping is disabled every beam stays enabled. Clearing keeps
124  // the capacity from previous updates, so the storage is only reallocated if the beam count grows.
125  const std::size_t num_beams = points.size();
126  beam_mask_.clear();
127  beam_mask_.reserve(num_beams);
128 
129  if (!do_beamskip_ || num_beams == 0) {
130  beam_mask_.resize(num_beams, static_cast<std::uint8_t>(true));
131  return;
132  }
133 
134  const auto unknown_space_occupancy_prob = static_cast<float>(1. / this->params_.max_laser_distance);
135  std::vector<std::size_t> obs_count(num_beams, 0);
136  std::size_t num_states = 0;
137 
138  for (const auto& state : states) {
139  ++num_states;
140  const auto transform = this->world_to_likelihood_field_transform_ * state;
141  const auto x_offset = transform.translation().x();
142  const auto y_offset = transform.translation().y();
143  const auto cos_theta = transform.so2().unit_complex().x();
144  const auto sin_theta = transform.so2().unit_complex().y();
145  for (std::size_t i = 0; i < num_beams; ++i) {
146  const auto& point = points[i];
147  // Transform the end point of the laser to the grid local coordinate system.
148  const auto x = point.first * cos_theta - point.second * sin_theta + x_offset;
149  const auto y = point.first * sin_theta + point.second * cos_theta + y_offset;
150  const auto pz = this->likelihood_field_.data_near(x, y).value_or(unknown_space_occupancy_prob);
151  // The likelihood field is monotonically decreasing in the distance to the nearest
152  // obstacle, so "distance < beam_skip_distance" is equivalent to "pz > threshold".
153  if (pz > likelihood_threshold_) {
154  ++obs_count[i];
155  }
156  }
157  }
158 
159  if (num_states == 0) {
160  beam_mask_.resize(num_beams, static_cast<std::uint8_t>(true));
161  return;
162  }
163 
164  std::size_t skipped = 0;
165  for (std::size_t i = 0; i < num_beams; ++i) {
166  const double ratio = static_cast<double>(obs_count[i]) / static_cast<double>(num_states);
167  beam_mask_.push_back(static_cast<std::uint8_t>(ratio > beam_skip_threshold_));
168  if (!beam_mask_.back()) {
169  ++skipped;
170  }
171  }
172 
173  // Safety fallback: if too many beams would be skipped, integrate all of them instead.
174  const double skipped_ratio = static_cast<double>(skipped) / static_cast<double>(num_beams);
175  if (skipped_ratio > beam_skip_error_threshold_) {
176  std::fill(beam_mask_.begin(), beam_mask_.end(), static_cast<std::uint8_t>(true));
177  }
178  }

197  [[nodiscard]] auto operator()(measurement_type&& points) const {
198  return [this, points = std::move(points)](const state_type& state) -> weight_type {
199  const auto transform = this->world_to_likelihood_field_transform_ * state;
200  const auto x_offset = transform.translation().x();
201  const auto y_offset = transform.translation().y();
202  const auto cos_theta = transform.so2().unit_complex().x();
203  const auto sin_theta = transform.so2().unit_complex().y();
204  const auto unknown_space_occupancy_prob = static_cast<float>(1. / this->params_.max_laser_distance);
205 
206  const auto beam_log_likelihood = [this, x_offset, y_offset, cos_theta, sin_theta,
207  unknown_space_occupancy_prob](const auto& tuple) {
208  const auto& [point, is_beam_used] = tuple;
209  if (!is_beam_used) {
210  return 0.0; // Neutral element of the log-likelihood sum below.
211  }
212  // Transform the end point of the laser to the grid local coordinate system.
213  // Not using Eigen/Sophus because they make the routine x10 slower.
214  // See `benchmark_likelihood_field_model.cpp` for reference.
215  const auto x = point.first * cos_theta - point.second * sin_theta + x_offset;
216  const auto y = point.first * sin_theta + point.second * cos_theta + y_offset;
217  const auto pz =
218  static_cast<double>(this->likelihood_field_.data_near(x, y).value_or(unknown_space_occupancy_prob));
219  return std::log(pz);
220  };
221 
222  // Beams masked out by prepare() contribute nothing to the weight. When prepare() was not called
223  // for this measurement (e.g. when the model is used standalone, outside the two pass update)
224  // every beam is integrated, reproducing the plain likelihood field prob behavior. The mask is
225  // borrowed from this sensor model and the fallback is a lazy view, so no memory is allocated.
226  if (this->beam_mask_.size() == points.size()) {
227  return std::exp(
228  ranges::accumulate(ranges::views::zip(points, this->beam_mask_), 0.0, std::plus{}, beam_log_likelihood));
229  }
230  return std::exp(ranges::accumulate(
231  ranges::views::zip(points, ranges::views::repeat(static_cast<std::uint8_t>(true))), 0.0, std::plus{},
232  beam_log_likelihood));
233  };
234  }

252  static float compute_likelihood_threshold(const param_type& params) {
253  assert(!params.do_beamskip || params.beam_skip_distance <= params.max_obstacle_distance);
254 
255  const auto profile = LikelihoodFieldModelBase<OccupancyGrid>::make_likelihood_profile(params);
256  const double squared_distance = params.beam_skip_distance * params.beam_skip_distance;
257  return static_cast<float>(profile(squared_distance));
